CREATE TABLE departamentos (
	id_departamento SERIAL PRIMARY KEY,
	nombre_departamento VARCHAR(100) NOT NULL
);

CREATE TABLE rol (
	id_rol SERIAL PRIMARY KEY,
	nombre_rol VARCHAR(50) NOT NULL
);

CREATE TABLE usuarios (
	id_usuario SERIAL PRIMARY KEY,
	nombre VARCHAR(50),
	apellido VARCHAR(50),
	correo VARCHAR(100) NOT NULL UNIQUE,
	nombre_usuario VARCHAR(25),
	password_hash VARCHAR(255) NOT NULL,
	id_departamento INT REFERENCES departamentos(id_departamento),
	id_rol INT REFERENCES rol(id_rol)
);

CREATE TABLE tickets (
	id_ticket SERIAL PRIMARY KEY,
	id_usuario INT REFERENCES usuarios(id_usuario),
	id_tecnico INT REFERENCES usuarios(id_usuario),
	fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	descripcion TEXT NOT NULL,
	prioridad VARCHAR(20) NOT NULL CHECK (prioridad IN ('Baja','Media','Alta','Critica')),
	estado VARCHAR(20) NOT NULL CHECK (estado IN ('Abierto','En Progreso','Resuelto'))
);

CREATE TABLE bitacora (
	id_bitacora SERIAL PRIMARY KEY,
	id_ticket INT REFERENCES tickets(id_ticket),
	id_usuario INT REFERENCES usuarios(id_usuario),
	comentario TEXT NOT NULL,
	estado_anterio VARCHAR(20),
	estado_nuevo VARCHAR(20) NOT NULL,
	fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

SELECT 
	d.nombre_departamento,
	COUNT(t.id_ticket) AS total_criticos
FROM tickets t 
INNER JOIN usuarios u ON t.id_usuario = u.id_usuario
INNER JOIN departamentos d ON u.id_departamento = d.id_departamento
WHERE t.prioridad = 'Critica'
GROUP BY d.nombre_departamento
HAVING COUNT(t.id_ticket) > 5;
